// Copy into an isolated Unity 6 project's Assets/Editor, then run
// -batchmode -nographics -executeMethod RebuildDefaultMaterial.Run.
using System;
using UnityEditor;
using UnityEngine;

public static class RebuildDefaultMaterial
{
    public static void Run()
    {
        const string path = "Packages/com.shanflyer.ui-effects/Shaders/UIAdditive.mat";
        Material fresh = null;
        try
        {
            var shader = Shader.Find("ShanFlyer/UI Effects/Additive");
            var existing = AssetDatabase.LoadAssetAtPath<Material>(path);
            if (!shader || !existing) throw new InvalidOperationException("Package shader/material is missing.");
            string guid = AssetDatabase.AssetPathToGUID(path);
            fresh = new Material(shader) { name = "UIAdditive" };
            EditorUtility.CopySerialized(fresh, existing);
            EditorUtility.SetDirty(existing);
            AssetDatabase.SaveAssets();
            if (guid != AssetDatabase.AssetPathToGUID(path)) throw new InvalidOperationException("Material GUID changed.");
            Debug.Log("Default material rebuilt from shader defaults; GUID=" + guid);
            EditorApplication.Exit(0);
        }
        catch (Exception error)
        {
            Debug.LogException(error);
            EditorApplication.Exit(1);
        }
        finally { if (fresh) UnityEngine.Object.DestroyImmediate(fresh); }
    }
}
